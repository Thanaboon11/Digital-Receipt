        function addToList() {

            const p = document.getElementById('nodePrice');
            if (!t.value || !p.value) return; [cite: 17]

            myItems.push({ id: Date.now(), title: t.value, price: parseFloat(p.value) }); [cite: 17]

        function updateDisplay() {

                    </div>

        }

        function removeItem(id) {

        function changePax(v) {

            if(res < 1) res = 1; [cite: 26]

        function showFinalReceipt() {

            const pax = parseInt(document.getElementById('paxValue').value) || 1; [cite: 28]

        async function takeScreenshot() {
            const actions = document.getElementById('actionButtons');
            actions.classList.add('hide-in-photo');
            const element = document.getElementById('receiptUI');
            try {
                const canvas = await html2canvas(element, {
                    scale: 3,
                    backgroundColor: "#ffffff",
                    borderRadius: 48,
                    useCORS: true
                });
                const link = document.createElement('a');
                link.href = canvas.toDataURL("image/png");
                link.download = `DaySplit-${Date.now()}.png`;
                link.click();
            } catch (err) {
                alert('การบันทึกขัดข้อง ลองแคปหน้าจอแทนนะครับ');
            } finally {
                actions.classList.remove('hide-in-photo');
            }
        }
