        function saveItem() {
            const t = document.getElementById('nodeTitle'), p = document.getElementById('nodePrice'), e = document.getElementById('editId');
            if (!t.value || !p.value) return;
            const dateStr = new Date().toLocaleDateString('th-TH', {day:'2-digit', month:'short', year:'2-digit'});
            const itemData = { title: t.value, price: parseFloat(p.value), date: dateStr };
            if (e.value) { db.child(e.value).update(itemData); e.value = ''; document.getElementById('btnSave').innerText = '+'; }
            else { db.push(itemData); }
            t.value = ''; p.value = ''; t.focus();
            document.getElementById('receiptUI').classList.add('hidden');
        }

        db.on('value', snap => {
            const data = snap.val() || {};
            const items = Object.entries(data).map(([id, val]) => ({ id, ...val }));
            window.allItems = items;
            const container = document.getElementById('dailyContainer');
            if (items.length === 0) { container.innerHTML = ''; return; }
            const groups = items.reduce((acc, i) => { (acc[i.date] = acc[i.date] || []).push(i); return acc; }, {});
            container.innerHTML = Object.keys(groups).sort((a,b)=>b.localeCompare(a)).map(date => `
                <div class="space-y-2">
                    <div class="text-[9px] font-black text-slate-400 ml-2 uppercase tracking-widest">${date}</div>
                    ${groups[date].map(i => `
                        <div onclick="prepEdit('${i.id}')" class="bg-white p-4 rounded-xl shadow-sm flex justify-between items-center cursor-pointer active:scale-95 transition-all">
                            <div><div class="font-bold text-slate-700 text-sm">${i.title}</div><div class="text-blue-600 font-black">฿${i.price.toLocaleString(undefined,{minimumFractionDigits:2})}</div></div>
                            <button onclick="event.stopPropagation(); deleteItem('${i.id}')" class="text-slate-200 text-xl px-2">✕</button>
                        </div>
                    `).join('')}
                </div>
            `).join('');
        });

        function prepEdit(id) {
            const item = window.allItems.find(x => x.id === id);
            document.getElementById('nodeTitle').value = item.title;
            document.getElementById('nodePrice').value = item.price;
            document.getElementById('editId').value = id;
            document.getElementById('btnSave').innerText = '✓';
            window.scrollTo({top: 0, behavior: 'smooth'});
        }

        function deleteItem(id) { if(confirm('ลบรายการนี้?')) db.child(id).remove(); }
        function clearData() { if(confirm('ล้างข้อมูลทั้งหมด?')) db.remove(); }
        function changePax(v) { let i = document.getElementById('paxValue'); i.value = Math.max(1, parseInt(i.value) + v); }

        function showReceipt() {
            if (!window.allItems || window.allItems.length === 0) return alert('ไม่มีรายการ');
            const p = parseInt(document.getElementById('paxValue').value) || 1, 
                  t = window.allItems.reduce((s, i) => s + i.price, 0);
            
            document.getElementById('receiptRows').innerHTML = window.allItems.map(i => `
                <div class="flex justify-between items-center">
                    <span class="text-sm font-bold text-slate-500">${i.title}</span>
                    <span class="text-sm font-black text-slate-800">฿${i.price.toLocaleString(undefined,{minimumFractionDigits:2})}</span>
                </div>`).join('');
            
            document.getElementById('resSum').innerText = '฿' + t.toLocaleString(undefined,{minimumFractionDigits:2});
            document.getElementById('resFinal').innerText = (t/p).toLocaleString(undefined,{minimumFractionDigits:2});
            
            const now = new Date();
            const timeStr = now.toLocaleTimeString('th-TH', { hour: '2-digit', minute: '2-digit', second: '2-digit' });
            const dateStr = now.toLocaleDateString('th-TH', { day: 'numeric', month: 'long', year: 'numeric' });
            document.getElementById('resDate').innerText = `${dateStr} | ${timeStr} น.`;
            
            document.getElementById('receiptUI').classList.remove('hidden');
            document.getElementById('receiptUI').scrollIntoView({behavior: 'smooth'});
        }

        async function takeScreenshot() {
            const b = document.getElementById('actionButtons'); b.classList.add('hide-in-photo');
            const canvas = await html2canvas(document.querySelector('.standard-bill'), { scale: 3 });
            const a = document.createElement('a'); a.href = canvas.toDataURL(); a.download = `Bill.png`; a.click();
            b.classList.remove('hide-in-photo');
        }
