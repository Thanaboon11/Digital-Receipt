    <script>
        let myItems = [];

        function addToList() {
            const t = document.getElementById('nodeTitle');
            const p = document.getElementById('nodePrice');
            if (!t.value || !p.value) return;

            myItems.push({ id: Date.now(), title: t.value, price: parseFloat(p.value) });
            t.value = ''; p.value = ''; t.focus();
            updateDisplay();
            document.getElementById('receiptUI').classList.add('hidden');
        }

        function updateDisplay() {
            const body = document.getElementById('dataTable');
            const noData = document.getElementById('noData');
            document.getElementById('count').innerText = myItems.length;

            if (myItems.length === 0) {
                body.innerHTML = '';
                noData.style.display = 'block';
                return;
            }

            noData.style.display = 'none';
            body.innerHTML = myItems.map((item, index) => `
                <tr class="border-b border-slate-50">
                    <td class="p-5">
                        <div class="item-text text-lg">${item.title}</div>
                        <div class="text-[10px] text-slate-400 font-bold uppercase">Item #${index+1}</div>
                    </td>
                    <td class="p-5 text-right">
                        <span class="price-text text-xl">${item.price.toLocaleString(undefined,{minimumFractionDigits:2})}</span>
                    </td>
                    <td class="p-5 text-center action-col">
                        <button onclick="removeItem(${item.id})" class="text-slate-300 hover:text-red-500 text-xl">🗑</button>
                    </td>
                </tr>
            `).join('');
        }

        function removeItem(id) {
            myItems = myItems.filter(x => x.id !== id);
            updateDisplay();
            document.getElementById('receiptUI').classList.add('hidden');
        }

        function changePax(v) {
            const inp = document.getElementById('paxValue');
            let res = parseInt(inp.value) + v;
            if(res < 1) res = 1;
            inp.value = res;
        }

        function showFinalReceipt() {
            if (myItems.length === 0) return alert('กรุณากรอกข้อมูลก่อนครับ');
            const pax = parseInt(document.getElementById('paxValue').value) || 1;
            const total = myItems.reduce((s, i) => s + i.price, 0);
            
            document.getElementById('receiptRows').innerHTML = myItems.map(i => `
                <div class="flex justify-between font-bold text-slate-600 italic">
                    <span>${i.title}</span>
                    <span class="text-slate-900">${i.price.toLocaleString(undefined,{minimumFractionDigits:2})} ฿</span>
                </div>
            `).join('');
            
            document.getElementById('resSum').innerText = total.toLocaleString(undefined,{minimumFractionDigits:2}) + ' ฿';
            document.getElementById('resPax').innerText = pax + ' คน';
            document.getElementById('resFinal').innerText = (total / pax).toLocaleString(undefined,{minimumFractionDigits:2});
            document.getElementById('resDate').innerText = new Date().toLocaleString('th-TH');

            const ui = document.getElementById('receiptUI');
            ui.classList.remove('hidden');
            ui.scrollIntoView({ behavior: 'smooth' });
        }

        // ฟังก์ชันถ่ายภาพใบเสร็จ
        async function takeScreenshot() {
            const btn = document.querySelector('button[onclick="takeScreenshot()"]');
            const actions = document.getElementById('actionButtons');
            
            // ซ่อนปุ่มก่อนถ่าย
            actions.classList.add('hide-in-photo');
            
            const element = document.getElementById('receiptUI');
            
            try {
                const canvas = await html2canvas(element, {
                    scale: 3, // ความชัดระดับ 3x
                    backgroundColor: "#ffffff",
                    borderRadius: 32,
                    useCORS: true,
                    logging: false
                });
                
                // สร้างลิงก์ดาวน์โหลด
                const image = canvas.toDataURL("image/png");
                const link = document.createElement('a');
                link.href = image;
                link.download = `DaySplit-Receipt-${Date.now()}.png`;
                link.click();
                
                alert('บันทึกรูปภาพเรียบร้อยแล้ว!');
            } catch (err) {
                alert('เกิดข้อผิดพลาดในการบันทึกรูปภาพ');
                console.error(err);
            } finally {
                // เอากลุ่มปุ่มกลับคืนมา
                actions.classList.remove('hide-in-photo');
            }
        }
    </script>
